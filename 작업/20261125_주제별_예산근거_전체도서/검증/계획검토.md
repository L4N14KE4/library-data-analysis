확인 코드 (프로젝트 폴더에서 실행, 계산명세 핵심 수치를 보고하려는 것이 아니라 표준편차 계산 방식에 따라 값이 갈리는지만 본 것):

```python
import pandas as pd, numpy as np, statistics, math
from fractions import Fraction
d = pd.read_csv("P02_장서목록_대출횟수 original.csv", encoding="utf-8-sig", dtype=str,
                keep_default_na=False, usecols=["대출횟수"])
x = [int(v.strip()) for v in d["대출횟수"]]; a = np.array(x, dtype=float); n = len(x); S = sum(x)
mu = S / n
for name, s in [("np", float(np.std(a, ddof=1))), ("pd", float(pd.Series(a).std())),
                ("stat", statistics.stdev(x)),
                ("exact", math.sqrt(float((Fraction(sum(v*v for v in x)) - Fraction(S*S, n)) / (n-1))))]:
    print(name, repr(s), repr(mu - 3*s), repr(mu + 3*s))
```
